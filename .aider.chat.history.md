Jarvis - голосовой помощник для Fedora Linux

def process_command(cmd: str):
    """Обработка команды"""
    cmd_lower = cmd.lower().strip()

    if "выход" in cmd_lower or "закрой" in cmd_lower or "выключи" in cmd_lower:
        return False

    elif "помощь" in cmd_lower or "команды" in cmd_lower:
        list_commands()
    elif "слушай" in cmd_lower or "ввод" in cmd_lower:
        voice_input_text()
    elif "файлы" in cmd_lower or "создай" in cmd_lower or "открой" in cmd_lower:
        file_menu()
    elif "громкость" in cmd_lower:
        set_volume_from_command(cmd_lower)
    elif "открой" in cmd_lower or "запусти" in cmd_lower:
        launch_app(cmd_lower)
    elif "что открыто" in cmd_lower or "окна" in cmd_lower:
        list_open_windows()
    elif "сделай скриншот" in cmd_lower or "скриншот" in cmd_lower:
        take_screenshot()
    elif "затиши" in cmd_lower or "убавь" in cmd_lower:
        adjust_volume(cmd_lower, -10)
    elif "прибавь" in cmd_lower or "громче" in cmd_lower:
        adjust_volume(cmd_lower, 10)
    elif "поставь на паузу" in cmd_lower or "пауза" in cmd_lower:
        media_pause()
    else:
        # Передаём в распознаваемый текст активного окна
        logger.info(f"Команда не распознана: {cmd}")
        speak("Команда не распознана. Введите 'помощь' для списка.")
    return True


def list_commands():
    commands = [
        "слушай / ввод - голосовой ввод текста в активное окно",
        "файлы / создай файл / открой - управление файлами",
        "громче / тише / громкость 50 - управление звуком",
        "открой браузер / запусти firefox - запуск приложений",

import os
import time
import logging
import speech_recognition as sr
from config import TMP_DIR, SYSTEM_LANGUAGE

logger = logging.getLogger(__name__)

# Глобальный распознаватель (один экземпляр)
recognizer = sr.Recognizer()
microphone = sr.Microphone()


def setup_audio():
    """Настройка микрофона"""
    with microphone as source:
        logger.info("Настройка уровней аудио...")
        recognizer.adjust_for_ambient_noise(source, duration=1)


def listen_microphone(timeout=5, phrase_time_limit=10):
    """Слушать микрофон один раз"""
    setup_audio()
    try:
        logger.info("Слушаю...")
        audio = recognizer.listen(
            microphone,
            timeout=timeout,
            phrase_time_limit=phrase_time_limit
        )
        return audio
    except sr.WaitTimeoutError:
        logger.debug("Время прослушивания истекло")
        return None
    except sr.UnknownValueError:
        logger.warning("Речь не распознана")
        return None
    except sr.RequestError as e:
        logger.error(f"Ошибка сервиса распознавания: {e}")
        return None


def recognize_command(language: str = SYSTEM_LANGUAGE):
    """Распознать команду с микрофона"""
    audio = listen_microphone()
    if audio is None:
        return None

    try:
        command = recognizer.recognize_google(
            audio,
            language=language
        )
        logger.info(f"Распознано: {command}")
        return command
    except sr.UnknownValueError:
        logger.warning("Речь не распознана (google)")
        return None
    except sr.RequestError as e:
        logger.error(f"Ошибка Google Speech Recognition: {e}")
        return None


def listen_for_wake_word(timeout=5):
    """Слушать слово пробуждения"""
    setup_audio()
    try:
        audio = recognizer.listen(microphone, timeout=timeout)
        word = recognizer.recognize_google(audio, language="ru")
        if WAKE_WORD.lower() in word.lower():
            logger.info(f"Слово пробуждения обнаружено: {WAKE_WORD}")
            return None  # Следующий вызов распознает команду
        return None
    except sr.WaitTimeoutError:
        return None
    except Exception as e:
        logger.error(f"Ошибка проверки пробуждения: {e}")
        return None


def set_wake_word(word: str):
    global WAKE_WORD
    WAKE_WORD = word

import os
import logging
import subprocess
import sys
from config import TTS_ENGINE, TTS_LANGUAGE, TMP_DIR

logger = logging.getLogger(__name__)


def speak(text: str):
    """Произнести текст голосом"""
    if not text:
        return

    text = text.strip()
    if not text:
        return

    logger.debug(f"🔊 Говорю: {text}")

    try:
        if TTS_ENGINE == "gtts":
            tts_google(text)
        elif TTS_ENGINE == "espeak":
            tts_espeak(text)
        else:
            tts_espeak(text)
    except Exception as e:
        logger.error(f"Ошибка синтеза речи: {e}")
        print(f"[JARVIS] Ошибка озвучки: {e}")


def tts_google(text: str):
    """Google TTS через gTTS"""
    try:
        from gtts import gTTS
        filename = os.path.join(TMP_DIR, "response.mp3")
        tts = gTTS(text=text, lang=TTS_LANGUAGE, slow=False)
        tts.save(filename)
        subprocess.run(["paplay", filename], check=True)
    except ImportError:
        logger.warning("gTTS не установлен, переключаемся на espeak")
        tts_espeak(text)
    except Exception as e:
        logger.error(f"Ошибка gTTS: {e}")


def tts_espeak(text: str):
    """espeak (предустановлен на Fedora)"""
    # Экранирование кавычек и спецсимволов
    safe_text = text.replace('"', '\\"')
    cmd = f'espeak -v ru -p 10 -s 150 "{safe_text}"'
    try:
        subprocess.run(cmd, shell=True, check=True)
    except subprocess.CalledProcessError as e:
        logger.error(f"Ошибка espeak: {e}")
        # Резервный вариант

            shell=True
        )

#!/usr/bin/env python3
import logging
import subprocess
import pyautogui

logger = logging.getLogger(__name__)


def adjust_volume(cmd: str, delta: int):
    """Изменить громкость на +/- delta%"""
    try:
        current = get_volume()
        new_vol = min(100, max(0, current + delta))
        cmd_line = f"pactl set-sink-volume @DEFAULT_SINK@ {new_vol}%"
        subprocess.run(cmd_line, shell=True, check=True)
        logger.info(f"Громкость установлена: {new_vol}%")
        return new_vol
    except Exception as e:
        logger.error(f"Ошибка изменения громкости: {e}")
        return current


def get_volume():
    """Текущая громкость"""
    try:
        out = subprocess.run(
            ["pactl", "get-sink-volume", "@DEFAULT_SINK@"],
            capture_output=True, text=True, check=True
        )
        for line in out.stdout.splitlines():
            if "%" in line:
                return int(line.split("%")[0].split()[-1])
    except Exception as e:
        logger.error(f"Ошибка получения громкости: {e}")
    return 50


def set_volume_from_command(cmd: str):
    """Установить громкость по числу"""
    import re
    nums = re.findall(r"\d+", cmd)
    if nums:
        vol = min(100, max(0, int(nums[0])))
        subprocess.run(
            f"pactl set-sink-volume @DEFAULT_SINK@ {vol}%",
            shell=True, check=True
        )
        logger.info(f"Громкость: {vol}%")
        return vol
    return get_volume()


def launch_app(cmd: str):
    """Запустить приложение"""
    app_map = {
        "браузер": "firefox",
        "хром": "chromium-browser",
        "chrome": "google-chrome",
        "терминал": "gnome-terminal",
        "текстовый редактор": "gedit",
        "vlc": "vlc",
        "mail": "thunderbird",
    }
    # ищем ключ по подстроке
    matched = None
    for key in app_map:
        if key in cmd:
            matched = app_map[key]
            break
    if matched:
        subprocess.Popen([matched], shell=False)
        logger.info(f"Запущено: {matched}")
    else:
        logger.warning(f"Приложение не найдено для команды: {cmd}")
        return None
    return matched


def list_open_windows():
    """Список открытых окон"""
    try:
        out = subprocess.run(

            capture_output=True, text=True, check=True

        windows = out.stdout.strip().split("\n")

            if w.strip():

#!/usr/bin/env python3
import logging
import pyautogui
import time

from recognition import recognize_command
from tts import speak

logger = logging.getLogger(__name__)


def voice_input_text():
    """Режим голосового ввода: надиктовать текст, закрыть окно ввода, вставить"""
    print("\n=== 🎙 ГОЛОСОВОЙ ВВОД ТЕКСТА ===")
    print("Скажите команду или текст. Скажите 'завершить', чтобы выйти.")
    collect = []
    while True:
        text = recognize_command()
        if not text:
            continue

        if "завершить" in text.lower() or "хватит" in text.lower():
            break

        print(f"Вы: {text}")
        collect.append(text)

    full_text = " ".join(collect)
    if not full_text:
        print("Текст не собран.")
        return

    print(f"📝 Вставляю: {full_text}")
    # Пытаемся найти активное окно редактирования
    # Снимаем фокус и возвращаем (помогает вставить)
    time.sleep(0.5)
    # Вставляем текст
    pyautogui.write(full_text, interval=0.03)
    print("Текст вставлен в активное окно.")
    speak(f"Вставил: {full_text[:100]}")


def paste_last_dctated(paste_buffer):
    """Вставить последний продиктованный текст"""
    if paste_buffer:
        pyautogui.write(paste_buffer[-1], interval=0.03)
        return True
    return False

def ask_llm(context: list) -> dict:
    """
    Отправить контекст в OpenRouter и получить ответ в формате JSON.
    Возвращает dict с intent, args, answer.
    """
    if not OPENROUTER_API_KEY:
        logger.warning("API ключ OpenRouter не задан — используем заглушку")

### Общая структура проекта

Проект представляет собой голосового помощника **Jarvis** для Linux (Fedora), работающий в командной строке.

---

### Модули

#### `jarvis/config.py`
Глобальная конфигурация проекта. Хранит настройки:
- API ключи для LLM (`OPENROUTER_API_KEY`)
- Параметры синтеза речи (TTS_ENGINE, TTS_LANGUAGE)
- Слово пробуждения, громкость, пути к файлам
- Системную подсказку для LLM (`SYSTEM_PROMPT`)

#### `jarvis/jarvis.py`
Основной модуль, точка входа (`main()`). Включает:
- Вывод приветственного баннера
- Обработку команд (`process_command()`)
- Слушание микрофона (с проверкой слова пробуждения)
- Отображение списка команд

#### `jarvis/file_ops.py`
Операции с файлами: создание, открытие, удаление, меню.

#### `jarvis/llm.py`
Взаимодействие с LLM через OpenRouter. Содержит `ask_llm()` (отправка запроса) и `understand()` (распознавание намерения).

#### `jarvis/pc_control.py`
Управление ПК: громкость (`get_volume`, `adjust_volume`), запуск приложений, список окон, скриншоты, пауза медиа.

#### `jarvis/recognition.py`
Распознавание речи с микрофона: `recognize_microphone`, `recognize_google`, `recognize_whisper`, работа со словом пробуждения.

#### `jarvis/tts.py`
Синтез речи: `speak`, `tts_google`, `tts_espeak`.

#### `jarvis/voice_input.py`
Голосовой ввод текста в активное окно.

#### `context.py`
Хранение и управление контекстом диалога.

#### `profile.py`
Менеджер профиля пользователя (загрузка, сохранение, обновление).

#### `requirements.txt`
Зависимости: `SpeechRecognition`, `pyaudio`, `whisper`, `gTTS`, `pyttsx3`.
